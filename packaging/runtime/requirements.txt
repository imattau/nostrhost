# Pinned Python runtime deps bundled into the `nostrhost-runtime` package.
#
# build-package's `runtime` kind runs:
#   pip download --only-binary=:all: --platform manylinux_2_17_x86_64 \
#                --python-version 3.11 --implementation cp --abi cp311 -r this
# and stages the wheels into /usr/share/nostrhost/wheels; postinst installs
# them offline into /opt/nostrhost/venv (created with --system-site-packages
# so the dist-packages debs stay visible).
#
# These packages are not in Debian bookworm (or only in an incompatible
# version — bookworm ships pydantic v1), so they previously had to be
# pip-installed by hand on the target. Keep this list aligned with the
# component uv.locks:
#   - libs/nostrhost-auth   -> nostr-sdk
#   - libs/nostrhost-policy -> bech32, coincurve, pydantic
# Transitive pins (pydantic-core, typing-extensions, annotated-types,
# typing-inspection) are pinned here too so the wheel set is reproducible.
#
# IMPORTANT: when any pin here changes, bump the `nostrhost-runtime` version
# in packaging/packages.yml — apt does not upgrade a same-version rebuild.
nostr-sdk==0.45.1
bech32==1.2.0
coincurve==21.0.0
pydantic==2.13.5
pydantic-core==2.46.5
typing-extensions==4.16.0
annotated-types==0.8.0
typing-inspection==0.4.4
# The daemons + control-plane clients use websockets.sync.client (>=12.0);
